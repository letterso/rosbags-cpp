cmake_minimum_required(VERSION 3.16)
project(rosbags_install_consumer LANGUAGES CXX)
enable_testing()
if(PREEXISTING_DEPENDENCIES)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(LZ4 REQUIRED IMPORTED_TARGET liblz4)
  pkg_check_modules(ZSTD REQUIRED IMPORTED_TARGET libzstd)
  add_library(LZ4::lz4 INTERFACE IMPORTED)
  set_property(TARGET LZ4::lz4 PROPERTY INTERFACE_LINK_LIBRARIES PkgConfig::LZ4)
  add_library(zstd::libzstd INTERFACE IMPORTED)
  set_property(TARGET zstd::libzstd PROPERTY INTERFACE_LINK_LIBRARIES PkgConfig::ZSTD)
endif()
find_package(rosbags_cpp CONFIG REQUIRED)
find_package(rosbags_cpp CONFIG REQUIRED)
if(PREEXISTING_DEPENDENCIES)
  get_target_property(lz4_links LZ4::lz4 INTERFACE_LINK_LIBRARIES)
  get_target_property(zstd_links zstd::libzstd INTERFACE_LINK_LIBRARIES)
  if(NOT lz4_links STREQUAL "PkgConfig::LZ4" OR NOT zstd_links STREQUAL "PkgConfig::ZSTD")
    message(FATAL_ERROR "Existing compression dependency targets were overwritten")
  endif()
endif()
add_executable(example main.cpp)
target_link_libraries(example PRIVATE rosbags_cpp::rosbags_cpp)
if(WITH_GENERATOR)
  set(generator_args)
  if(HOST_GENERATOR)
    list(APPEND generator_args GENERATOR "${HOST_GENERATOR}")
  endif()
  set(MESSAGE_PROFILE application CACHE STRING "Generated profile")
  rosbags_generate_messages(custom PROFILE "${MESSAGE_PROFILE}" INPUTS definitions ${generator_args})
  target_link_libraries(example PRIVATE custom)
  target_compile_definitions(example PRIVATE WITH_GENERATOR)
endif()
add_test(NAME example COMMAND example)
